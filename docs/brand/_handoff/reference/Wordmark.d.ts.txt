/* REFERENCE ONLY — the prop contract for the Wordmark primitive. */

/** Type-set brand lockup (no logo asset exists). Two systems: the bare bracketed [UN] parent anchor (suffix={null}), and [UN] {Suffix} sub-brand lockups. Anton bracket + Space Mono suffix at 1.27× the bracket size (cap-height matched). */
export interface WordmarkProps{
  /** sm 16 / md 20 / lg 36, or a px number. */
  size?:'sm'|'md'|'lg'|number;
  /** Division suffix in Space Mono, sentence case: 'Hinged' (default), 'Bound', 'Limited', 'Scripted', 'Cut'. Pass null for the bare [UN] parent anchor (System A). */
  suffix?:'Hinged'|'Bound'|'Limited'|'Scripted'|'Cut'|string|null;
  /** Optional mono sub line under the lockup (e.g. 'SINGLES SOCIAL CLUB'). Null by default. */
  sub?:string|null;
  /** Sub line color. Defaults to the division's own accent. */
  accent?:'hinged'|'bound'|'limited'|'scripted'|'cut'|'shop';
  /** Suffix in editorial serif italic lowercase — campaign headlines and deck openers only. */
  editorial?:boolean;
  /** Suffix in mono ALL CAPS — large physical goods only (screen print, cap backs, flags). */
  caps?:boolean;
  inverse?:boolean;
}
export declare function Wordmark(props:WordmarkProps):JSX.Element;
